// src/ui/TaskList.tsx (due date parts)
type Task = { id: number; title: string; done: number; due_date: string | null };

function DueInput({ task, onSaved }: { task: Task; onSaved: (t: Task) => void }) {
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);
  async function save(v: string) {
    setSaving(true); setError(null);
    try {
      const r = await fetch(`/tasks/${task.id}`, { method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ due_date: v || null }) });
      const body = await r.json();
      if (!r.ok) setError(body.error ?? "Could not save");
      else onSaved(body);
    } catch { setError("Network error"); }
    finally { setSaving(false); }
  }
  return (<>
    <input type="date" aria-label="Due date" value={task.due_date ?? ""}
      disabled={saving} onChange={e => save(e.target.value)} />
    {error && <span role="alert" className="error">{error}</span>}
  </>);
}

// src/api/tasks.ts
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
const validDate = (v: unknown) => {
  if (v === null || v === undefined) return true;
  if (typeof v !== "string" || !DATE_RE.test(v)) return false;
  const d = new Date(v + "T00:00:00Z");
  return !isNaN(+d) && d.toISOString().slice(0, 10) === v;
};
const BAD = { error: "due_date must be YYYY-MM-DD" };

router.post("/tasks", (req, res) => {
  const { title, due_date = null } = req.body;
  if (!title) return res.status(400).json({ error: "title required" });
  if (!validDate(due_date)) return res.status(400).json(BAD);
  const created_at = new Date().toISOString();
  const id = db.insert("tasks", { title, done: 0, due_date, created_at });
  res.status(201).json({ id, title, done: 0, due_date, created_at });
});

router.patch("/tasks/:id", (req, res) => {
  const patch: Partial<Pick<Task, "title" | "done" | "due_date">> = {};
  for (const k of ["title", "done", "due_date"] as const)
    if (k in req.body) (patch as any)[k] = req.body[k];
  if ("due_date" in patch && !validDate(patch.due_date)) return res.status(400).json(BAD);
  res.status(200).json(db.update(req.params.id, patch));
});
